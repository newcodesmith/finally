**Create test/docker-compose.test.yml:**
```yaml
services:
  app:
    build:
      context: ..
      dockerfile: Dockerfile
    ports:
      - "8000:8000"
    environment:
      - LLM_MOCK=true
      - DB_PATH=/app/db/test.db
    volumes:
      - test-data:/app/db
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/api/health"]
      interval: 5s
      timeout: 3s
      retries: 30
      start_period: 30s

volumes:
  test-data:
```

Note: The Playwright tests will run on the host (not in a container) against http://localhost:8000 after the app container is healthy. This is simpler and allows better debugging than running Playwright inside a container. The test workflow is:
1. `cd test && docker compose -f docker-compose.test.yml up -d --build --wait`
2. `cd test && npx playwright install --with-deps chromium`
3. `cd test && npx playwright test`
4. `cd test && docker compose -f docker-compose.test.yml down -v`

Install deps: `cd test && npm install`
